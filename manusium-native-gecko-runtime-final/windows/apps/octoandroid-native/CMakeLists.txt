cmake_minimum_required(VERSION 3.24)
project(OctoAndroidManagerNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(OctoAndroidManager WIN32 main.cpp)
target_compile_definitions(OctoAndroidManager PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(OctoAndroidManager PRIVATE comctl32 shell32)
if(MSVC)
  target_compile_options(OctoAndroidManager PRIVATE /W4 /EHsc)
endif()
